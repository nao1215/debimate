CREATE TABLE orders (
  id      bigint PRIMARY KEY,
  shop_id int    NOT NULL,
  amount  int    NOT NULL
) WITH (autovacuum_enabled = off);
CREATE TABLE
INSERT INTO orders
  SELECT g,
         CASE WHEN g % 100 = 0 THEN 2 + (g / 100) % 1000 ELSE 1 END,
         g % 1000
  FROM generate_series(1, 1000000) AS g;
