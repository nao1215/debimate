<script>
const b64 = b => btoa(String.fromCharCode(...new Uint8Array(b))).replace(/\+/g,'-').replace(/\//g,'_').replace(/=+$/,'');
async function register(rpId) {
  const c = await navigator.credentials.create({publicKey: {
    rp: {id: rpId, name: "lab"}, user: {id: new Uint8Array(16), name: "alice", displayName: "alice"},
    challenge: crypto.getRandomValues(new Uint8Array(32)),
    pubKeyCredParams: [{type: "public-key", alg: -7}],
    authenticatorSelection: {residentKey: "required", userVerification: "required"}}});
  return {id: c.id, clientData: new TextDecoder().decode(c.response.clientDataJSON)};
}
async function login(rpId) {
  try {
    const a = await navigator.credentials.get({publicKey: {rpId, userVerification: "required",
      challenge: crypto.getRandomValues(new Uint8Array(32))}});
    const ad = new Uint8Array(a.response.authenticatorData);
    return {ok: true, clientData: new TextDecoder().decode(a.response.clientDataJSON),
            signCount: new DataView(ad.buffer).getUint32(33), flags: ad[32]};
  } catch (e) { return {ok: false, error: e.name}; }
}
</script>
