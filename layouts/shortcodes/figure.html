{{/* PaperMod の figure ショートコードの上書き。 本家は src をそのまま出力するため、page bundle 内の画像を相対パスで 指定すると、週報一覧 (/weekly/) のように別 URL のページで .Content を 展開した時にパスが壊れる。ここでは page bundle のリソースを RelPermalink (絶対パス) へ解決してから出力する。 リソースが見つからない場合は、本家と同じく src をそのまま使う。 */}} {{- $src := .Get "src" -}} {{- $u := urls.Parse $src -}} {{- if not $u.IsAbs -}} {{- $path := strings.TrimPrefix "./" $u.Path -}} {{- with or (.Page.Resources.Get $path) (resources.Get $path) -}} {{- $src = .RelPermalink -}} {{- end -}} {{- end -}} {{- if .Get "link" -}} {{- end }} {{- if .Get "link" }}{{ end -}} {{- if or (or (.Get "title") (.Get "caption")) (.Get "attr") -}} {{- end }}
