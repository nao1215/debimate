{{- /* Home-only SVG, regenerated from the title at build time. */ -}} {{- $title := .Title | plainify | htmlUnescape -}} {{- $parts := split $title " - " -}} {{- $heading := (index $parts 0) | truncate 54 -}} {{- $lines := findRE `.{1,18}` $heading -}} {{- $subtitle := delimit (after 1 $parts) " - " | truncate 52 -}} {{- $size := 56 -}} {{- if gt (len $lines) 3 }}{{ $size = 40 }}{{ $lines = findRE `.{1,22}` $heading }}{{ end -}} {{- $text := "" -}} {{- range $i, $line := $lines -}} {{- $text = printf "%s%s" $text (add 175 (mul $i (add $size 10))) $size ($line | htmlEscape) -}} {{- end -}} {{- $y := add 187 (mul (len $lines) (add $size 10)) -}} {{- range $i, $line := findRE `.{1,26}` $subtitle -}} {{- $text = printf "%s%s" $text (add $y (mul $i 44)) ($line | htmlEscape) -}} {{- end -}} {{- $svg := printf `` (.CurrentSection.Title | htmlEscape) $text -}} {{- $image := resources.FromString (printf "note-cards/%s.svg" (md5 (printf "%s%s" .RelPermalink $svg))) $svg -}} {{- return $image.RelPermalink -}}
