{{- /* Inline home cover lettering: render with the document, without a lazy image request. */ -}} {{- /* ページか、ページを持たない外部記事の dict（title・label）を受け取る。 */ -}} {{- $title := "" -}}{{- $label := "" -}} {{- if reflect.IsMap . }}{{ $title = .title }}{{ $label = .label }}{{ else }}{{ $title = .Title }}{{ $label = .CurrentSection.Title }}{{ end -}} {{- $title = $title | plainify | htmlUnescape -}} {{- $parts := split $title " - " -}} {{- /* truncate は HTML エスケープした文字列を返すので、戻してから行ごとに 1 回だけエスケープする。 */ -}} {{- $heading := (index $parts 0) | truncate 54 | htmlUnescape -}} {{- $lines := partial "notes/wrap.html" (dict "text" $heading "width" 18) -}} {{- $subtitle := delimit (after 1 $parts) " - " | truncate 52 | htmlUnescape -}} {{- $size := 44 -}} {{- if gt (len $lines) 3 }}{{ $size = 36 }}{{ $lines = partial "notes/wrap.html" (dict "text" $heading "width" 22) }}{{ end -}} {{- $text := "" -}} {{- range $i, $line := $lines -}} {{- $text = printf "%s%s" $text (add 175 (mul $i (add $size 10))) $size ($line | htmlEscape) -}} {{- end -}} {{- $y := add 187 (mul (len $lines) (add $size 10)) -}} {{- range $i, $line := partial "notes/wrap.html" (dict "text" $subtitle "width" 26) -}} {{- $text = printf "%s%s" $text (add $y (mul $i 44)) ($line | htmlEscape) -}} {{- end -}} {{- $svg := printf `` ($label | htmlEscape) $text -}} {{- return ($svg | safeHTML) -}}
